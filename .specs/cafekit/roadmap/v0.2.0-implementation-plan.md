const https = require('https');

async function checkLatestVersion() {
  return new Promise((resolve, reject) => {
    https.get('https://registry.npmjs.org/@haposoft/cafekit-spec/latest', (res) => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => {
        const json = JSON.parse(data);
        resolve(json.version);
      });
    }).on('error', reject);
  });
}

### Morning (1 hour): Uninstall Command

**Task 6.1: Create uninstall.js**
```javascript
const fs = require('fs');
const readline = require('readline');

async function uninstallCommand() {
  const files = findCafeKitFiles(); // Search for files with CafeKit signature
  
  if (files.length === 0) {
    console.log('ℹ️  No CafeKit files found.');
    return;
  }
  
  console.log(`Found ${files.length} CafeKit Spec files:`);
  files.forEach(file => console.log(`- ${file}`));
  
  // Confirm
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
  });
  
  rl.question('\n⚠️  Remove these files? (y/N): ', (answer) => {
    if (answer.toLowerCase() === 'y') {
      files.forEach(file => {
        fs.unlinkSync(file);
        console.log(`🗑️  Removed: ${file}`);
      });
      console.log('\n✅ CafeKit Spec uninstalled.');
    } else {
      console.log('❌ Cancelled.');
    }
    rl.close();
  });
}

module.exports = uninstallCommand;
```

**Verify:** Install → uninstall → Files removed

### Afternoon (1 hour): Config File Support

**Task 7.1: Create config-reader.js**
```javascript
const fs = require('fs');
const yaml = require('js-yaml');

function readConfig() {
  const configPath = './.cafekit.yml';
  
  if (!fs.existsSync(configPath)) {
    return {}; // No config file
  }
  
  try {
    const fileContents = fs.readFileSync(configPath, 'utf8');
    const config = yaml.load(fileContents);
    return config;
  } catch (e) {
    console.error('❌ Error reading .cafekit.yml:', e.message);
    return {};
  }
}

module.exports = readConfig;
```

**Task 7.2: Update install.js**
